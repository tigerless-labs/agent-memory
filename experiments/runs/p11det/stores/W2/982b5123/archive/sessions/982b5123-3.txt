class S3Uploader:
    def __init__(self, bucket_name):
        self.s3 = boto3.client("s3")
        self.bucket_name = bucket_name

    def upload_directory(self, directory, checksum_file_path):
        try:
            for root, dirs, files in os.walk(directory):
                for file in files:
                    file_path = os.path.join(root, file)
                    relative_path = os.path.relpath(file_path, directory)
                    s3_key = os.path.join(os.path.basename(directory), relative_path)
                    self.s3.upload_file(file_path, self.bucket_name, s3_key, ExtraArgs={
                        "Metadata": {
                            "checksum": self._get_checksum(relative_path, checksum_file_path)
                        }
                    })
        except Exception as e:
            raise S3OperationException(f"Failed to upload directory to S3: {e}")

    def _get_checksum(self, relative_path, checksum_file_path):
        with open(checksum_file_path, "r") as f:
            for line in f.readlines():
                parts = line.strip().split(",")
                if parts[0] == relative_path:
                    return parts[1]
        return ""

class S3Downloader:
    def __init__(self, bucket_name):
        self.s3 = boto3.client("s3")
        self.bucket_name = bucket_name

    def download_directory(self, directory, checksum_file_path):
        try:
            for root, dirs, files in os.walk(directory):
                for file in files:
                    file_path = os.path.join(root, file)
                    relative_path = os.path.relpath(file_path, directory)
                    s3_key = os.path.join(os.path.basename(directory), relative_path)
                    s3_object = self.s3.get_object(Bucket=self.bucket_name, Key=s3_key)
                    with open(file_path, "wb") as f:
                        f.write(s3_object["Body"].read())
                    checksum = s3_object["Metadata"].get("checksum", "")
                    expected_checksum = self._get_checksum(relative_path, checksum_file_path)
                    if checksum != expected_checksum:
                        raise S3OperationException("Checksum mismatch")
        except Exception as e:
            raise S3OperationException(f"Failed to download directory from S3: {e}")

    def _get_checksum(self, relative_path, checksum_file_path):
        with open(checksum_file_path, "r") as f:
            for line in f.readlines():
                parts = line.strip().split(",")
                if parts[0] == relative_path:
                    return parts[1]
        return ""
